## Machine-level notes

- Source: raw capture + R10-014 live re-verify (`decompile_function` ≡ raw CF).
- **ABI:** `__thiscall` — `ECX` = mission/format context (`this`, TFID at `+0x30`); **5 stack args**; **`RET 0x14`**; returns **bool** in `AL`.
- **Role:** format **one mission-dialog token value** into caller out-buffer, switched on `tokenDesc->type` (byte 0..6).
- Sole caller `FUN_00579fb0` walks a token list + format string: parse segment → call this → on success append value & advance; on fail wrap raw token text in marker words.
- Dualed callees: `Character_GetRaceDisplayName_Inferred` (`0x00521800`), `Character_GetClassDisplayName_Inferred` (`0x00521900`).
- Prefer bytes when decompiler invents `unaff_*` destinations — out buffer is always stack arg 1.

Mission_FormatDialogTokenValue_Inferred(
    void *context, char *outBuf, char *formatOrTokenText,
    TokenDesc *tokenDesc, void *objectOrFloatSlot, void *mapOrCharacter)
{
  void *tlsRing = FUN_007a69d0();
  switch (tokenDesc->type) {
  case 0: {
    float value = 0.0f;
    if (mapOrCharacter == NULL ||
        !TFID_EqualsObjectId((char*)mapOrCharacter + 0x160, (char*)context + 0x30)) {
      if (!CVOGMap_LookupVariable(tokenDesc->varId, &value, mapOrCharacter))
        return 0;

    }
    sprintf(outBuf, formatOrTokenText, (double)value);
    return 1;
  }
  case 1:
    if (objectOrFloatSlot == NULL) return 0;

    strcpy(outBuf, Object_GetName_vtbl160(objectOrFloatSlot));
    return 1;
  case 2:
    if (objectOrFloatSlot == NULL) return 0;
    strcpy(outBuf, FUN_007a6de0(
        Character_GetClassDisplayName_Inferred(objectOrFloatSlot), -1));
    return 1;
  case 3:
    if (objectOrFloatSlot == NULL) return 0;
    strcpy(outBuf, FUN_007a6de0(
        Character_GetRaceDisplayName_Inferred(objectOrFloatSlot), -1));
    return 1;
  case 4:
    if (mapOrCharacter != NULL && Character_IsCharacter_vtbl210(mapOrCharacter)) {
      void *ch = Character_GetSelf_vtbl210(mapOrCharacter, 0, -1);
      strcpy(outBuf, FUN_007a6de0(Character_GetRaceDisplayName_Inferred(ch), -1));
      return 1;
    }
    strcpy(outBuf, FUN_007a6de0("!Non-Character Faction not implemented yet!", -1));
    return 1;
  case 5:
    if (mapOrCharacter == NULL) return 0;
    if (Character_IsCharacter_vtbl210(mapOrCharacter)) {
      void *ch = Character_GetSelf_vtbl210(mapOrCharacter, 0);
      strcpy(outBuf, FUN_007a6de0(Object_GetName_vtbl160(ch), -1));
      return 1;
    }
    strcpy(outBuf, FUN_007a6de0(*(char**)((int*)mapOrCharacter)[0x2a] + 0x40, -1));
    return 1;

    if (mapOrCharacter != NULL && Character_IsCharacter_vtbl210(mapOrCharacter)) {
      void *ch = Character_GetSelf_vtbl210(mapOrCharacter, 0, -1);
      strcpy(outBuf, FUN_007a6de0(Character_GetClassDisplayName_Inferred(ch), -1));
      return 0; // machine: write then fail — parent will not advance token
    }
    *outBuf = '\0';
    return 0;
  }
  return 0;
}
