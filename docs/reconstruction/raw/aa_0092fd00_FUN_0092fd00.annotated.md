## Machine-level notes

- **ABI is dual-register custom, not cdecl void:** `EDI` = client, `EAX` = missionId, return `AL`.

```
{
  Character* ch = *(client + 0xe98);
  if (!ch) return 0;

  if (missionId != 0) {
    void* node = CNDHash_LookupByKey(*(ch + 0x548), missionId);
    if (!node) return 0;
    // thiscall: ECX=node, stack=ch  (FUN_0059dc50 gate)
    if (!ObjectiveNode_CanBindToCharacter(node, ch)) return 0;
    // fall into bind
  } else {
    Tracker* tr = *(client + 0x4d0);
    if (*(tr + 0x10) == 0) return 1;   // already idle — no UI work
  }

  // bind / clear via tracker
  Tracker* tr = *(client + 0x4d0);
  char ok = Tracker_SetActiveMissionObjective(tr /*ECX*/, missionId /*EAX*/); // FUN_0092a590
  if (ok && *(client + 0x107c)) {
    auto* ui = *(client + 0x107c);
    if (ui->vtbl[+0x3d8]())   // visible/live
      ui->vtbl[+0x448]();     // refresh/hide-toggle pair used elsewhere as refresh
  }
  return ok;
}
```
