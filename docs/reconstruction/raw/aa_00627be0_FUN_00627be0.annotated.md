## Machine-level notes

- **Leaf:** no callees; pure field presence scan.
- **ABI:** `ECX` = skill runtime (sealed at Reevaluate / FUN_0054fec0 call sites).
- **Return:** `*(uint*)(skill+0xc) | presenceBits` in EAX.
- **Consumer pattern (not this body):** store to `skill+0x624`; if `skill+0x620 != 0` then `| 4` and optional `skill+0x620 |= 0x100000` when `skill+0x148==0`.
- Decimal decompiler literals: `200` → `0xc8`, `300` → `0x12c`.
- Paired short groups: base at `+0x78` / `+0x84` with second bank at `base+0x18` (`psVar2[0xc]` = +24 bytes = +12 shorts).

// ECX = skill runtime after ranked elements materialize.

uint __fastcall Skill_ComputeEffectPresenceFlags_Inferred(int pSkill)
{
  uint flags = 0;
  int i;
  short *ps;
  float *pf;

  // bit 0x1 — primary short bank (+0x60, 6 slots)
  for (i = 0, ps = (short *)(pSkill + 0x60); i < 6; i++, ps++) {
    if (*ps != 0) { flags = 1; break; }
  }
  // bit 0x8 — secondary short bank (+0x6c, 6)
  for (i = 0, ps = (short *)(pSkill + 0x6c); i < 6; i++, ps++) {
    if (*ps != 0) { flags |= 8; break; }
  }
  // bit 0x80 — float bank (+0xa8, 6)
  for (i = 0, pf = (float *)(pSkill + 0xa8); i < 6; i++, pf++) {
    if (*pf != 0.0f) { flags |= 0x80; break; }
  }
  // bit 0x100 — paired shorts (+0x78 / +0x90)
  for (i = 0, ps = (short *)(pSkill + 0x78); i < 6; i++, ps++) {
    if (ps[0xc] != 0 || *ps != 0) { flags |= 0x100; break; }
  }
  // bit 0x200 — paired shorts (+0x84 / +0x9c); fall into scalar block
  for (i = 0, ps = (short *)(pSkill + 0x84); i < 6; i++, ps++) {
    if (ps[0xc] != 0 || *ps != 0) { flags |= 0x200; break; }
  }

  // scalar / misc presence (shared tail)
  if (/* any of misc float group */) flags |= 0x400;
  if (/* any i16 +0xd0..+0xd8 */)     flags |= 0x40;
  if (f32[+0x38] || f32[+0x13c])      flags |= 0x20;
  if (char[+0xda])                    flags |= 0x10;
  if (f32[+0x3c])                     flags |= 2;
  if (f32[+0x40])                     flags |= 0x8000;
  if (/* +0x10c..+0x118 group */)     flags |= 0x10000;
  if (f32[+0x124] || f32[+0x128])     flags |= 0x20000;
  if (f32[+0x12c] || f32[+0x130])     flags |= 0x400;   // second contribution to same bit
  if (i32[+0x134] && f32[+0x138])     flags |= 0x20000;

  return *(uint *)(pSkill + 0xc) | flags;
}
