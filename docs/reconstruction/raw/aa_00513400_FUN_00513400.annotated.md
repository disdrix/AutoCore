## Pseudocode (annotated; CF from raw + disasm)

```c
uint32_t __thiscall Item_ValidateGadgetAttach_Inferred(Item* host, Gadget* gadget)
{
  // gadgetType via cloned-object ac-chain + template +0x4c4
  int gadgetType = *(*(*(gadget+4)+4) + gadget + 0xac)->+0x3c -> +0x4c4;
  int* hostComp = *(int**)(host + 0xa8);

  if (gadgetType != hostComp[0x38/4]) {
    if (gadgetType == 0x44) {
      if (*(short*)(*(hostComp + 0x3c/4) + 0x3f4) == 10) goto ok;
    } else if (gadgetType == 0x46 &&
               *(short*)(*(hostComp + 0x3c/4) + 0x3f4) == 11) {
      goto ok;
    }
    return 2;  // "wrong type of gadget"
  }
ok:
  short used = host->vtbl[+0x8c](host);
  int   cap  = host->vtbl[+0x80](host);
  if (used == cap)
    return 1;  // "No gadget slots are available."

  uint32_t gadgetId = *(*(gadget ac-chain) + 0x34);
  for (uint32_t i = 0; i < (uint32_t)host->vtbl[+0x80](host); ++i) {
    uint16_t slotId = (uint16_t)host->vtbl[+0x7c](host, i);
    if (slotId == (uint16_t)gadgetId)
      return 3;  // "already on the item, no duplicates"
  }
  return 0;    // allow
}
```
