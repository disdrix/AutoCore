## Purpose

Periodic MissionTracker target refresh: clear owned pointer vector at `host+0x11c`, reload poll timer to **4.0f**, resolve nearest world target(s) from character origin, bind primary hit into a tracker node (TFID copy), or fall back to secondary multi-target scan.

// AL  = success

{

    if (!client || !client->character /*+0xE98*/ || !*(client+0xE04))
        return 0;

    // EDI = host for clear helper
    Tracker_ClearOwnedPtrVectorAt11c_Inferred();   // FUN_0092a3d0
    *(float*)(host + 0x14) = 4.0f;                 // DAT_00aaa690

    Character* ch = *(Character**)(client + 0xE98);
    if (*(uint8_t*)(ch + 0x4F1) != 0 || !*(client + 0xD34))
        return 0;

    // this-adjust character; vtbl +0x1A0
    float* origin = Character_GetOrigin_vtbl1A0(ch);

    void*  bestObj    = NULL;
    float  bestDistSq = 1.0e9f;                    // DAT_00aaaaf8

    // EAX=host (filter ctx), ESI=&bestObj
    // stack: origin, &bestDistSq
    if (FUN_0092af80(origin, &bestDistSq) /*multi-list, includes 0092adf0*/) {
        if (bestObj != NULL) {
            if (*(uint8_t*)(host + 0x0B) != 0) {
                int type = *(*(int**)(bestObj + 0xA8) + 0x38/4); // clonebase+0x38
                if (type != 0x12 && type != 0x0E)
                    *(uint8_t*)(host + 0x0B) = 0;
            }
            // ECX=bestObj, stack=host
            void* node = FUN_0092a440(bestObj, host);
            if (node == NULL)
                return 1;                          // success, no TFID store
            *(uint8_t*)(host + 0x08) = 0;
            memcpy(node, (uint8_t*)bestObj + 0x160, 16); // TFID_16
            return 1;
        }

    // secondary fill path

        return 1;

    *(uint8_t*)(host + 0x08) = 0;
    return 0;
}
