// cdecl; plain RET; callers may push 6th residual
void GuardedVector_AssignLinearRangeAt(
    GuardedVectorIteratorPair* out,
    const uint32_t* first,
    const uint32_t* last,
    GuardedVectorHeader* c,
    uint32_t dest_idx)
{
  for (; first != last; ++first) {
    uint32_t page = dest_idx >> 2;
    uint32_t sub  = dest_idx - (page * 4);
    if (c->capacity <= page) {
      page -= c->capacity;
    }
    uint32_t* slot = c->pages[page] + sub;
    if (slot != first) {
      *slot = *first;
    }
    dest_idx++;
  }
  out->index = dest_idx;
  out->container = c;
}
