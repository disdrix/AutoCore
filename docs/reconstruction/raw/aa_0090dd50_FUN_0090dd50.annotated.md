## Purpose

Recompute camera **distance-band** fields from integer **mode** at `cam+0x60c`, then **clamp** current distance slots into the new `[min,max]`.

Callers typically write `mode` into `DAT_00d1b778+0x60c` (saved camera cookie from special events) then invoke this with the camera object in **EAX**.

## Algorithm — SEALED

```
mode = cam->mode@+0x60c

if mode == 0:
  min = 2.0
  max = (DAT_00d1b26c != 0) ? 16.0 : 11.0
  rate = (max - min) * (1.0 / 16.0)     // ≡ (max-min)/16
  if DAT_00d1b6d8 && *(DAT_00d1b6d8 + 0x6b9):
    secMax = (DAT_00d1b26c != 0) ? 30.0 : 20.0
    span = secMax - 6.0
    cam->+0x52c = span * 0.5            // 7.0 or 12.0
    cam->+0x530 = span/16 + span*0.5    // 7.875 or 13.5
else if mode == 1:
  min = 6.0
  max = (DAT_00d1b26c != 0) ? 30.0 : 20.0
  rate = (max - min) / 16.0
else:  // mode >= 2 (or any non-0/1)
  min = 12.0
  max = 20.0
  cam->+0x52c = cam->+0x530 = 20.0
  rate = (max - min) / 16.0             // 0.5

// shared clamp of current slots into [min,max]
if cam->+0x530 > max:  cam->+0x530 = max
else if cam->+0x530 < min: cam->+0x530 = min

if cam->+0x52c > max:  cam->+0x52c = max; return
else if cam->+0x52c < min: cam->+0x52c = min
return
```

### Decompiler float compares

The decompiler expands `fcom`/`fucomip` pairs into verbose `a < b || a == b` / `a <= b && a != b` forms. Semantically these are ordinary **clamp to [min,max]** (inclusive). Clean plate uses clamp form; raw preserves decompiler compares.

### Mode-0 note

Mode 0 only **writes** `+0x52c/+0x530` when the local-player flag path runs. Otherwise those fields keep prior values and only get **clamped** at the end.

## Annotated pseudocode

```c
// cam in EAX
void ClientCamera_ApplyModeDistanceBand(void /* EAX = cam */)
{
  int mode = *(int *)(cam + 0x60c);

  if (mode == 0) {
    *(float *)(cam + 0x524) = 2.0f;                         // DAT_00af9154
    float maxD = (DAT_00d1b26c != 0) ? 16.0f : 11.0f;       // 915c / 9158
    float invSpan = g_flOne / 16.0f;                        // 1/af9150
    *(float *)(cam + 0x528) = maxD;
    *(float *)(cam + 0x534) = (maxD - 2.0f) * invSpan;

    if (DAT_00d1b6d8 && *(char *)(DAT_00d1b6d8 + 0x6b9)) {
      float secMax = (DAT_00d1b26c != 0) ? 30.0f : 20.0f;  // 9168 / 9164
      float span = secMax - 6.0f;                           // - af9160
      float mid = span * 0.5f;                              // * a0f298
      *(float *)(cam + 0x52c) = mid;
      *(float *)(cam + 0x530) = span * invSpan + mid;
    }
  } else if (mode == 1) {
    *(float *)(cam + 0x524) = 6.0f;                         // af9160
    float maxD = (DAT_00d1b26c != 0) ? 30.0f : 20.0f;
    *(float *)(cam + 0x528) = maxD;
    *(float *)(cam + 0x534) = (maxD - 6.0f) / 16.0f;
  } else {
    *(float *)(cam + 0x524) = 12.0f;                        // af916c
    *(float *)(cam + 0x528) = 20.0f;                        // af9170
    *(float *)(cam + 0x530) = 20.0f;
    *(float *)(cam + 0x52c) = 20.0f;
    *(float *)(cam + 0x534) = (20.0f - 12.0f) / 16.0f;
  }

  // clamp +0x530 into [min,max]
  float minD = *(float *)(cam + 0x524);
  float maxD = *(float *)(cam + 0x528);
  float cur  = *(float *)(cam + 0x530);
  if (cur > maxD) *(float *)(cam + 0x530) = maxD;
  else if (cur < minD) *(float *)(cam + 0x530) = minD;

  // clamp +0x52c into [min,max]
  cur = *(float *)(cam + 0x52c);
  if (cur > maxD) { *(float *)(cam + 0x52c) = maxD; return; }
  if (cur < minD) *(float *)(cam + 0x52c) = minD;
}
```
