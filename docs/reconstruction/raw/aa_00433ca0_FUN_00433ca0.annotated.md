## Pseudocode (annotated)

```c
// for (i = start; i < end; ++i) {
//   dest = container->buf + i*0xC;
//   if (dest != NULL) {
//     copy 3 dwords from src + i*0xC
//   }
// }
void BitStream_HuffmanAlphabet_CopyRange(/*EDI*/ Cont12* container,
                                         /*ESI*/ const uint8_t* src,
                                         /*EAX*/ uint32_t endIdx,
                                         /*EDX*/ uint32_t startIdx)
{
  if (startIdx >= endIdx) return;
  uint32_t off = startIdx * 0xCu;
  uint32_t n = endIdx - startIdx;
  do {
    uint32_t* dest = (uint32_t*)(*(uint8_t**)((uint8_t*)container + 8) + off);
    if (dest != 0) {
      const uint32_t* s = (const uint32_t*)(src + off);
      dest[0] = s[0];
      dest[1] = s[1];
      dest[2] = s[2];
    }
    off += 0xCu;
    n -= 1;
  } while (n != 0);
}
```
