## Purpose (sealed structural)

Register an **object** into a **listing context** (manager in ECX):

1. Validate 64-bit COID at object `+0x160/+0x164` (throw `E_INVALIDARG` + log tag `"VOG_DEBUG_STOP"` if invalid).
2. Gate manager mode flags (`this+0xc` / `this+0xd`) against object mode char at `+0x168`.
3. Reject if already listed (`FUN_004e77d0` → soft return `0` + `"object %I64d (%d) is already listed"`).
4. Resolve list bucket key via `vtbl+0x208` (force key `0x30`) or type dword at `*(obj+0xa8)+0x38`.
5. Ensure hash bucket entry in `this+0x18` (`CNDHash_LookupByKey` / alloc `0x2c` + insert).
6. Allocate primary list handle via `FUN_004bc680` → store `obj+0xdc`.
7. Type-special secondary handles (`+0xd0` / mission notify) for types `0x12`, `0xe` (+ linked check), `0x14`.
8. Optional handles `+0xd4` (`FUN_005130e0(1)`) and `+0xd8` (flag bit5 at `+0x17c`) with extra `CVOGReaction_FailMissionNotify`.

**Not** a pure debug-stop helper — `"VOG_DEBUG_STOP"` is only the log tag on the invalid-COID throw path.

## Control flow map

```
invalid COID? → log VOG_DEBUG_STOP; throw 0x80070057
mode gate fail? → throw 0x80070057
already listed? → log; return 0
key = vtbl+0x208 ? 0x30 : type@meta+0x38
hash miss? → new(0x2c)+ctor+insert or throw 0x8007000E
primary FUN_004bc680 → +0xdc
type 0x12 | (0xe & linked+0x2ac==0): FUN_004bc680 → +0xd0 (hr<0 throw); notify
type 0x14 fallthrough: notify
else: skip notify
FUN_005130e0(1)? FUN_004bc680 → +0xd4
flag bit5? FUN_004bc680 → +0xd8; notify
return 0
```

## Annotated pseudocode

```c
// Object_RegisterListed @ 0x004bc180
// ECX = ListingContext* mgr; stack = Object* obj; ret 4

int32_t __thiscall Object_RegisterListed(int mgr, int *obj)
{
  int *o = obj;
  int coid_hi = obj[0x59];          // +0x164
  // SEH frame ...

  // COID must be > 0 as signed 64-bit (hi/lo)
  if ((coid_hi < 1) && ((coid_hi < 0) || (obj[0x58] == 0))) {
    FUN_007a4480(0, "VOG_DEBUG_STOP");
    throw HRESULT(0x80070057);      // E_INVALIDARG
  }

  // Mode: obj+0x168 selects which manager enable byte is required
  if ((char)obj[0x5a] == 0) {
    if (*(char *)(mgr + 0xd) == 0) throw HRESULT(0x80070057);
  } else if (*(char *)(mgr + 0xc) == 0) {
    throw HRESULT(0x80070057);
  }

  // Already listed?
  if (FUN_004e77d0(obj[0x58], coid_hi, obj, 0) != 0) {
    FUN_007a4480(1, "object %I64d (%d) is already listed",
                 o[0x58], o[0x59], *(int *)(o[0x2a] + 0x34));
    return 0;
  }

  // Bucket key
  uint key = ((**(code **)(*o + 0x208))() == 0)
             ? *(uint *)(o[0x2a] + 0x38)
             : 0x30;

  void *bucket = CNDHash_LookupByKey(*(void **)(mgr + 0x18), key);
  if (bucket == NULL) {
    void *node = operator_new(0x2c);
    int inst = (node == NULL) ? 0 : FUN_004bcb40(/*this=node*/);
    if (inst == 0) throw HRESULT(0x8007000E); // E_OUTOFMEMORY
    FUN_004bd1d0(*(int *)(o[0x2a] + 0x38), inst, 0);
  }

  int handle;
  FUN_004bc680(o, &handle);
  o[0x37] = handle;                 // +0xdc primary

  int type = *(int *)(o[0x2a] + 0x38);
  if (type == 0x12 ||
      (type == 0xe &&
       (**(code **)(*o + 0x1d4))() != 0 &&
       *(char *)((**(code **)(*o + 0x1d4))() + 0x2ac) == 0)) {
    int hr = FUN_004bc680(o, &handle);
    if (hr < 0) throw hr;
    o[0x34] = handle;               // +0xd0
    CVOGReaction_FailMissionNotify(&obj);
  } else if (type == 0x14) {
    CVOGReaction_FailMissionNotify(&obj);
  }

  if (FUN_005130e0(1)) {
    int hr = FUN_004bc680(o, &handle);
    if (hr < 0) throw hr;
    o[0x35] = handle;               // +0xd4
  }

  if (((uint)o[0x5f] >> 5 & 1) != 0) {  // +0x17c bit 5
    int hr = FUN_004bc680(o, &handle);
    if (hr < 0) throw hr;
    o[0x36] = handle;               // +0xd8
    CVOGReaction_FailMissionNotify(&obj);
  }
  return 0;
}
```
