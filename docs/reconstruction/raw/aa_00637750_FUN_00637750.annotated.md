## Pseudocode (annotated)

```c
// TNL special-message unpack for opcode 0x804D (MapInstanceListResponse).
// Wire (BitStream over in->data): u8 count; count × {u64, u16, u16, flag, flag}
// Rebuild: ByteBuffer data prefix {u32 0x804D, u32 count, Entry* heapArray}
// Entry size 0x18: u64 + u16z + u16z + bool + bool + pad

undefined4 * TNL_ByteBuffer_UnpackData(undefined4 *out, int *inBuf)
{
  // SEH setup …
  // stackHeader.tag = 0x804D
  FUN_0042b3a0(/*this=stack BitStream*/, inBuf[3] /*data*/, inBuf[4] /*size*/);

  BitStream_readBits(/*this=bs*/, 8, &countU8);
  count = (uint)countU8;
  entryBytes = count * 0x18;
  stackHeader.count = count;
  stackHeader.entries = operator_new[](entryBytes);

  if (count != 0) {
    Entry *e = (Entry *)((char *)stackHeader.entries + 0); // fill via cursor at +0x0c
    // decompiler: puVar9 = entries + 0xC; fields via puVar9[-3..+1]
    for (; count; --count, e = (Entry *)((char *)e + 0x18)) {
      BitStream_readBits(0x40, &e->id64);     // +0x00 / +0x04
      BitStream_readBits(0x10, &tmp16);
      e->fieldA = (uint)tmp16;                // +0x08 (u16 zero-ext)
      BitStream_readBits(0x10, &tmp16);
      e->fieldB = (uint)tmp16 & 0xffff;       // +0x0c
      e->flag0 = BitStream_readFlag_inlined(); // +0x10
      e->flag1 = BitStream_readFlag_inlined(); // +0x11
    }
  }

  totalSize = entryBytes + 8; // tag+count mass; pointer still only 12B header on stack
  ByteBuffer *bb = operator_new(0x18);
  if (bb) {
    FUN_0042b250(bb);                 // ObjectObject base
    *bb = &PTR_LAB_009d7b00;        // ByteBuffer vtable
    bb->size = totalSize;             // +0x10
    bb->data = malloc(totalSize);     // +0x0c
    bb->ownsMemory = 1;               // +0x14
  }
  // memcpy(bb->data, &stackHeader, totalSize);  // prefix {tag,count,ptr} (+ over-read of stack for size>12)
  *out = bb;
  bb->refcount++;                     // +0x08

  // destroy stack BitStream; free buffer if owns
  FUN_0042b270(stackRelated);
  if (--inBuf->refcount == 0)
    inBuf->vtbl->release(inBuf);      // (**in)(+8)
  return out;
}
```
