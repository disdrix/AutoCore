- **ECX** = abstract stream*; **EDX** = dest RGB byte*; **`ret 0x14`** (5 stack dwords).
- Stack: image*, header* (bpp @ `header+0x10`), pixel count, row y, start x.
- Stream read slot: **vtbl +8** `(buf, size, count=1)`.
- SEH: `LAB_009bccb0`.
- Jump table @ `0x00434f10` / index bytes @ `0x00434f24` (index = bpp − 8).
- **16-bit residual fixed in annotation:** R = `(pix & 0x1f) << 3` (bytes have `and 0x1f`; decompiler dropped mask).
- 32bpp: RGB → dest (+3); alpha via `FUN_00433280(image, x+i, y, a)`.

// ECX=stream*, EDX=destRgb*, stack: image, header, count, y, x0
// ret 0x14
void DecodeUncompressedTgaRow(
    Stream *stream,          // ECX
    uint8_t *dest,           // EDX  (RGB8 output; 3 bytes/pixel for 15/16/24/32 paths)
    NDAssetImage *image,     // [ebp+8]  — 32bpp alpha only
    TgaHeader *header,       // [ebp+0xc] — bpp at +0x10
    int count,               // [ebp+0x10]
    int y,                   // [ebp+0x14]
    int x0)                  // [ebp+0x18]
{
  switch (header->bpp /* +0x10 */) {
  case 8:
    stream->vtbl->Read(stream, dest, count, 1);
    break;

  case 15:
  case 16:
    for (int i = 0; i < count; ++i) {
      uint16_t pix;
      stream->vtbl->Read(stream, &pix, 2, 1);
      dest[0] = (uint8_t)((pix & 0x1f) << 3);       // R  (bytes seal &0x1f)
      dest[1] = (uint8_t)((pix >> 2) & 0xf8);        // G  (5-bit mid)
      dest[2] = (uint8_t)((pix >> 7) & 0xf8);        // B
      dest += 3;
    }
    break;

  case 24:
    stream->vtbl->Read(stream, dest, count * 3, 1);
    break;

  case 32:
    for (int i = 0; i < count; ++i) {
      uint8_t bgra[4];
      stream->vtbl->Read(stream, bgra, 4, 1);
      dest[0] = bgra[0];
      dest[1] = bgra[1];
      dest[2] = bgra[2];
      dest += 3;
      FUN_00433280(image, /*x*/ x0 + i, /*y*/ y, /*a*/ bgra[3]);
    }
    break;

  default:
    break; // no-op (header already validated in LoadTGA)
  }
}
