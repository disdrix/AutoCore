## Pseudocode (annotated)

```c
void __thiscall Outpost_UpdateBeaconShareRatio(int thisOutpost /* ECX */)
{
  float total = g_flOne;                 // default if lookup fails / missing
  float factionBeacons = 0.0f;
  float oldRatio = *(float*)(thisOutpost + 0x24c);

  // Map var: global total beacons
  CVOGMap_LookupVariable(*(u32*)(thisOutpost + 0x1d0), &total, /*flags*/ 0);

  // Map var: beacons for current controlling faction
  int faction = *(int*)(thisOutpost + 0x234);
  CVOGMap_LookupVariable(
      *(u32*)(thisOutpost + 0xf0 + faction * 0x48),
      &factionBeacons,
      0);

  if (total <= g_flZero) {
    // Do NOT update +0x24c. Log only.
    FUN_007a4480(0, "Total beacons was %f on map %s",
                 (double)total,
                 /* map name */ mapObject(thisOutpost) + 0xb4);
    return;
  }

  float newRatio = factionBeacons / total;
  *(float*)(thisOutpost + 0x24c) = newRatio;   // always store when total > 0

  if (oldRatio != newRatio) {
    // Pre-recast cleanup for flag==0 bank entries (sibling leave-FUN)
    FUN_00606ff0(thisOutpost, faction);         // thiscall ECX=this, stack faction
    Outpost_CastSkillsForBeaconShare(thisOutpost, faction);
  }
}
```
