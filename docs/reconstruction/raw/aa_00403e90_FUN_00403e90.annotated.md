## Pseudocode (annotated)

```c
// __thiscall: ECX = x; stack map*; RET 4; void
// Node: left@0 parent@4 right@8 color@0x30 isnil@0x31 size 0x38
// Map:  head@+4; root = head->parent
void __thiscall StdTree_Lrotate_Isnil31_Inferred(Node* x, MapShell* map)
{
  Node* y = x->right;                 // [ECX+8]
  x->right = y->left;
  if (y->left->isnil == 0)            // +0x31
    y->left->parent = x;
  y->parent = x->parent;

  Node* head = map->myhead;           // *(map+4)
  if (x == head->parent) {            // x was root
    head->parent = y;
  } else if (x == x->parent->left) {
    x->parent->left = y;
  } else {
    x->parent->right = y;
  }
  y->left = x;
  x->parent = y;
}
```
