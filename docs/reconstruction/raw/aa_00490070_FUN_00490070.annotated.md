## Machine-level notes

- Source: raw capture + live re-decompile + `read_memory` entry + `DAT_00aaa630`.
- Prefer raw switch/gotos for ports; clean mirrors them.
- `DAT_00aaa630` = **FLT_MAX** (`0x7F7FFFFF`) — min-distance sentinel.
- Entry: `this` = Client* in ECX; gate views `+0xC0` / `+0xBC`.
- `param_3` unused in body.
- Selection: **max priority byte**, then **min** `fStack_3c` distance.
- Two enumerator passes: flag low-byte 0 then 1 via `FUN_004bae00`.
- Does **not** send UseObject (parent `aa_009247b0` does).

{
  // locals: bestPri(local_75/bVar2), bestDist(local_64), candDist(local_74),
  //         bestObj(local_68), enumFlag(local_6c), passes(local_5c=2), ...

  if (*(int*)(client+0xc0)==0 || *(int*)(client+0xbc)==0)
    return NULL;

  FUN_0075c340(screenXY, rayNear, rayDir); // GfxView this residual
  bestDist = candDist = DAT_00aaa630; // FLT_MAX
  bestObj = NULL;
  enumFlag &= ~0xFF; // low byte 0
  bestPri = 0;
  passes = 2;

  do {
    iter = 0;
    obj = FUN_004bae00(enumFlag, &iter);
    savedPri = bestPri;
    while (obj) {
      pri = 0;
      // gate: flags[0x5f] bit6; vtbl+0x1cc blob; blob+8; blob+0xe; dist vtbl+0x78(5)
      if (!eligible) goto next;

      dist = DAT_00aaa630;
      // probe fills dist (fStack_3c)

      flags = *(short*)(*(obj[0x2a]+0x3c)+0x3f2);
      if (flags bit14) { candDist=dist; pri=0x50; }
      if (flags bit7 && pri<0x47) { candDist=dist; pri=0x46; }

      switch (*(obj[0x2a]+0x38)) { // clone type
      case 1: case 3:
        // mission: world char at client+0xb8→+0xe8a0; 00524520 then 005245d0
        // owner TFID slots; flag 0x2000 → pri 0x14; mission → 0x3c
        break;
      case 10: case 0xc: case 0x10: case 0x1c:
        // parent obj[0x2b]: retarget + pri 0x5a else 0x46
        break;
      case 0xe:
        // if !filterSelfOwned skip when owner==local char; else pri 0x5a
        break;
      case 0x12:
        // self filter; dynamic_cast Creature; vtbl+0x198 / 0040b1b0 → 0x3c/0x32/0x64
        break;
      case 0x14: default:
        // self filter; pri 0x28
        break;
      case 0x16:
        // pri 100
        break;
      case 0x1e: case 0x36: case 0x38:
        goto next;
      }

      // max pri, then min dist
      if (pri == savedPri) {
        if (candDist < bestDist) { bestDist=candDist; bestObj=obj; savedPri=pri; }
      } else if (savedPri < pri) {
        bestObj=obj; savedPri=pri; /* dist already in cand */
      }
next:
      bestPri = savedPri;
      obj = FUN_004bae00(enumFlag, &iter);
      savedPri = bestPri;
    }
    passes--;
    enumFlag = (enumFlag & ~0xFF) | ((enumFlag & 0xFF) == 0);
    if (passes == 0) return bestObj;
  } while (1);
}
