## Pseudocode (annotated)

```c
// Custom: ESI=widget*, BL=selected, force@stack, RET 4
void UI_TabChrome_SetSelected_Inferred(char force /*stack*/)
{
  // unaff_ESI = widget; unaff_BL = selected
  if (force == 0 && (char)widget[0x13f /*+0x4FC*/] == selected)
    return;

  int *childA = (int *)widget[0x206 /*+0x818*/];

  if (selected == 0) {
    // --- unselect / collapse chrome ---
    if (childA) {
      // vtbl +0xF0 query; if style object: set scale = (+0x4FD)>>1 at +0xFC
      // else pack color via +0x164/+0x15C with half scale
      // vtbl +0x34C invalidate
    }
    if ((int *)widget[0x205 /*+0x814*/]) {
      // childB vtbl +0x3C8(push 1, push 0)  // hide companion
    }
    if ((char)widget[0x13f] == 0)
      goto store;
    // grow size: GetSize(+0x120); w += +0x80C; h += +0x810; SetSize(+0x110)
  } else {
    // --- select / expand chrome ---
    if (childA) {
      // same style path with full +0x4FD (no SHR)
    }
    if ((int *)widget[0x205]) {
      // childB vtbl +0x3C8(push 1, push 1)  // show companion
    }
    if ((char)widget[0x13f] == selected)
      goto store;
    // shrink size: GetSize; w -= +0x80C; h -= +0x810; SetSize
  }
  // SetSize via vtbl +0x110
store:
  *(char *)(widget + 0x13f) = selected;  // +0x4FC
}
```
