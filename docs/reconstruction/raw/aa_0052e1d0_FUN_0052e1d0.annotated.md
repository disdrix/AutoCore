## Semantics (annotated)

```text
need = Character_ComputeReverseEngineerRequiredLevel(character, item)

// Gates → 0:
//   item == null
//   configured mod slots (00599dd0 on def) < 1
//   character+0x6b4 > 0   (free / privileged → twin returns 1.0; this returns 0)

// Primary material cost:
//   for i in 0..slot_count-1:
//     cbid = *(uint32*)(def + 0x498 + i*4)
//     rank gate via object-map type 0x1a, clone+0x4c8
//     if rank <= 1 → contrib 0
//     else LookupCloneBase(cbid) → rank-1 (or -1 miss)
//     primary_cost += contrib

// base = ceil(primary_cost * (1/3.5))   // DAT_009cdff4

// Secondary recipe rows (item vtbl+0x60 count, vtbl+0x5c row → FUN_005097b0):
//   for each row: sum 5 CBID material costs; store per-row; accumulate total

// If primary_cost == 0 → return 0
// Else required = max(
//     base,
//     FUN_005206d0(row_count, total_cost) if total != -1,
//     FUN_005206d0(tier, row_cost[tier]) for tier 1..5 where row_cost != -1
// )
// return required
```

`FUN_005206d0(tier, cost)` = `ceil(cost * inv_mult[tier])` with inv_mult = 1/{3.5,3.5,2.8,2.1,1.4,0.7} — inverse of dualed `FUN_00520520` capacity scales. Meaning: minimum integer skill level whose `floor(skill*mult)` can cover `cost`.

## Decompiler artifacts corrected

1. Signature is **not** `void` / no-arg — thiscall character + stack item, RET 4, int EAX.
2. `unaff_EBX` / FPU CW `CONCAT22` — row cost accumulator + FPU control restore; not true unaffiliated state.
3. `param_2 = unaff_retaddr` mid-loop — decompiler noise; item pointer remains valid.
4. Free path: **return 0** (differs from twin free → **1.0**).

## Pseudocode (behavior-preserving, cleaned)

```c
// ECX=character*, stack item*, EAX=int, RET 4
int __thiscall Character_ComputeReverseEngineerRequiredLevel_Inferred(
    void* character, void* item)
{
  if (item == 0) return 0;
  void* link = *(void**)((char*)item + 0xa8);
  void* def  = link ? *(void**)((char*)link + 0x3c) : 0;
  int slots = ItemDef_CountConfiguredModSlots_Inferred(def);
  if (slots < 1) return 0;
  if (*(int*)((char*)character + 0x6b4) > 0) return 0;

  CVOGLootGenerator_GetOrCreate();

  int primary = 0;
  int row_cost[6];
  for (int t = 1; t < 6; ++t) row_cost[t] = -1;

  // primary CBIDs @ def+0x498
  for (int i = 0; i < slots; ++i) {
    uint32_t cbid = *(uint32_t*)((char*)def + 0x498 + i * 4);
    primary += MaterialCostContrib(cbid, /*use_clonebase*/1);
  }

  int saved_primary = primary;
  int required = (int)ceil((double)((float)primary * DAT_009cdff4)); // 1/3.5

  int total = primary;
  unsigned rows = item_vtbl_count_rows(item); // vtbl+0x60
  if (rows != 0) {
    for (unsigned r = 0; r < rows; ++r) {
      int rc = 0;
      uint16_t prefix = item_vtbl_get_row(item, r); // vtbl+0x5c
      int* recipe = FUN_005097b0(prefix);
      for (int s = 0; s < 5; ++s) {
        if (!recipe) break;
        uint32_t cbid = (uint32_t)recipe[s];
        if (cbid == 0xffffffffu) continue;
        rc += MaterialCostContribFromLiveMap(cbid);
      }
      total += rc;
      if (r < 6) row_cost[r] = rc;
      (void)ceil((double)((float)rc * DAT_009cdff4)); // FPU side path in image
    }
  }

  if (saved_primary == 0) return 0;

  if (total != -1) {
    int v = FUN_005206d0(rows, total);
    if (required < v) required = v;
  }
  for (int tier = 1; tier < 6; ++tier) {
    int c = row_cost[tier];
    if (c != -1) {
      int v = FUN_005206d0(tier, c);
      if (required < v) required = v;
    }
  }
  return required;
}
```
