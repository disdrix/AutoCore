- **cdecl** out-string constructor pattern: `(out*, src*) → out*`.
- SEH frame `LAB_009ab825`; EH state `1` only around append temp + copy.
- SSO: if `_Myres < 0x10` data at `src+4`, else `*(char**)(src+4)`.
- Last-char tests use byte compares `0x3A` / `0x2F` / `0x5C` (`:` `/` `\`).

- Nested `FUN_00416510` not dual-owned here.

MsvcBasicString* BasicString_CopyEnsureTrailingDirSep(
    MsvcBasicString* out,          // param_1 / ESI
    const MsvcBasicString* src)    // param_2
{
  // SEH LAB_009ab825
  if (src->_Mysize == 0) {
    basic_string_copy_ctor(out, src);
    return out;
  }

  const char* data = (src->_Myres < 0x10)
      ? (const char*)((const uint8_t*)src + 4)
      : *(const char* const*)((const uint8_t*)src + 4);
  char last = data[src->_Mysize - 1];

  if (last != ':' && last != '/' && last != '\\') {
    MsvcBasicString tmp; // 0x1c
    // EH=1
    FUN_00416510(&tmp, src, '/');   // copy + append '/'
    basic_string_copy_ctor(out, &tmp);
    // EH clear low byte
    basic_string_dtor(&tmp);
    return out;
  }

  basic_string_copy_ctor(out, src);
  return out;
}
