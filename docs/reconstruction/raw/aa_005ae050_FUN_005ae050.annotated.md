void __fastcall StdTree_Dec_Val12(StdTreeNode_Val12 **it)
{
  StdTreeNode_Val12 *node = *it;

  // End iterator (nil header): predecessor = rightmost = header->right
  if (node->isnil != 0) {
    *it = node->right;
    return;
  }

  // Left child exists → rightmost of left subtree
  StdTreeNode_Val12 *left = node->left;
  if (left->isnil == 0) {
    StdTreeNode_Val12 *p = left;
    while (p->right->isnil == 0) {
      p = p->right;
    }
    *it = p;
    return;
  }

  // Climb while coming from left
  StdTreeNode_Val12 *parent = node->parent;
  if (parent->isnil == 0) {
    while (parent->isnil == 0 && *it == parent->left) {
      *it = parent;
      parent = parent->parent;
    }
    if (parent->isnil == 0) {
      *it = parent;
    }
  }
}
