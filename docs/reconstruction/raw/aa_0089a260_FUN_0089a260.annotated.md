## Purpose

Shared **row binder** for the refine panel commodity list: filter by host mode, localize/format a display label with quantity and optional quality prefix, then insert into one of two list widgets (`host+0x568` vs `host+0x56c`) based on configured mod-slot count.

Sole caller: `FUN_0089bc30` (`UI_RefinePanel_RebuildCommodityList_Inferred`) — twice per rebuild (reverse + forward paths).

## Annotated control flow

  if (!entry) return;
  def = entry->def; // +0x3c
  if (!def) return;

  case 1: if (qty < 1) return; break;
  case 2: if (def->tier /*+0x4c8*/ != 0) return; break;
  case 3: if (def->tier != 1) return; break;
  case 4: if (def->tier != 2) return; break;
  case 6: if (def->tier != 3) return; break;
  case 7:
    if (!host->filterObj /*+0x530*/) return;
    FUN_0052d450(host->filterObj, &out);
    // map-find entry+0x34; if not present return
    break;
  default: break; // case 0/5 fall through
  }
  // double-check mode 1 qty
  if (host->filterMode == 1 && qty < 1) return;

  // zero many ~128B string buffers (quality / name parts)

  locale = FUN_007a69d0();
  // load localized name from def+0x92 (string id) via FUN_00402d50 + FUN_007a6de0
  // strtok first token of name on " "

  lang = locale->lang; // +0x2c
  if (lang == 1) {
    // French-style: use full name; skip EN/DE quality split
    format "[qty] name"  (or [0] when qty<1)
    goto insert;
  }

  if (lang == 0 || lang == 3) {
    // compare first token to Salvaged/Patched/Functional/Perfect
    // if none match → quality_known = false
  }
  if (lang == 2) {
    // load DE quality adjective tables (Verwertbar*/Aufbereitet*/Hochwertig*/Perfekt*)
    // compare first token; if match quality_known path
  }

  // if quality prefix recognized: rest of tokens joined into secondary name
  // sprintf label:
  //   quality_known ? "[%i] %s, %s" / "[0] %s, %s"
  //                 : "[%i] %s"     / "[0] %s"

insert:
  modSlots = FUN_00599dd0(/*def-driven count*/);
  if (modSlots < 1) {
    if (def->productKey /*+0x4d0*/ == -1) return;
    list = host->listA; // +0x568
  } else {
    list = host->listB; // +0x56c
  }
  list->vtbl_AddRow(+0x45c)(label, entry->id /*+0x34*/, ...);
  list->vtbl_Bind(+0x464)(entry->id, ..., stack residual);
}
