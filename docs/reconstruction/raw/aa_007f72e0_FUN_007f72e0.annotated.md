## Pseudocode (annotated)

```c
/* ActionMap_TrySetSlotBind — try write DIK/mod into ActionMap slot
   if not already customized. Leaf. RET 8. */

uint8_t ActionMap_TrySetSlotBind(
    /*AL*/  uint8_t slotIndex,
    /*BL*/  uint8_t bindLane,     // 0 = primary pair, nonzero = alt pair
    /*SI*/  uint16_t keyCode,     // typically DIK
    /*DX*/  uint16_t modCode,     // secondary / modifier DIK (0 unbound)
    /*stack0*/ uint8_t* actionMap,
    /*stack1*/ uint32_t mode)     // 0..3
{
  if (mode > 3) return 0;

  static const uint32_t kMax[4]   = {5, 6, 0x78, 0x14};
  static const uint32_t kFlag[4]  = {0x0C, 0x110, 0x248, 0x1AA8};
  static const uint32_t kKey0[4]  = {0x06, 0x10A, 0x242, 0x1AA2};
  static const uint32_t kMod0[4]  = {0x30, 0x134, 0x26C, 0x1ACC};
  static const uint32_t kKey1[4]  = {0x08, 0x10C, 0x244, 0x1AA4};
  static const uint32_t kMod1[4]  = {0x32, 0x136, 0x26E, 0x1ACE};
  static const uint32_t kClrA[4]  = {0x0B, 0x10F, 0x247, 0x1AA7};
  static const uint32_t kClrB[4]  = {0x0A, 0x10E, 0x246, 0x1AA6};

  if (slotIndex >= kMax[mode]) return 0;

  uint8_t* slot = actionMap + slotIndex * 0x34;
  if (slot[kFlag[mode]] != 0) return 0; // already customized

  if (bindLane == 0) {
    *(uint16_t*)(slot + kKey0[mode]) = keyCode;
    *(uint16_t*)(slot + kMod0[mode]) = modCode;
  } else {
    *(uint16_t*)(slot + kKey1[mode]) = keyCode;
    *(uint16_t*)(slot + kMod1[mode]) = modCode;
  }
  slot[kClrA[mode]] = 0;
  slot[kClrB[mode]] = 0;
  return 1;
}
```
