- **streamEnd gate:** loop requires `cursor + base < streamEnd` (`window+0x14`) — unlike ReadU8/ReadU32 which only check `available`.

## Pseudocode (register ABI restored)

```c
// ESI = StoStreamWindow* window
// EDI = std::string* out
// bare RET; returns 0 success / 0xffffffff fail
int32_t stoChunkStreamWindow_ReadCString(void)
{
  std::string::resize(out /*EDI*/, 0);

  if (window->cursor + window->base >= window->streamEnd)
    return -1;

  do {
    if (window->available < window->cursor + 1) {
      int32_t hr = stoChunkStreamWindow_Ensure(window, window->base + window->cursor);
      if (hr < 0) return -1;
      if (window->available < 1) return -1;
    }

    char c = *(char*)((uint8_t*)window + 0x18 + window->cursor);
    window->cursor += 1;

    if (c == '\0')
      return 0;   // NUL consumed; string holds payload only

    std::string::operator+=(out, c);
  } while (window->cursor + window->base < window->streamEnd);

  return -1;  // hit streamEnd without NUL
}
```
