## Pseudocode (annotated; dual-grid this restored)

```c
bool __thiscall Character_CountType1aCraftableBatches(
    void *self, void *obj, int *outBatches, char mode, char simpleFlag)
{
  void *cargoHolder;
  void *cargo;   // InventoryGrid*
  void *locker;  // InventoryGrid*
  int *blob;
  int primaryHave, secondaryHave, slotCbid, minHave, i, nSlots;
  char modeNot2;

  *outBatches = 0;

  cargoHolder = *(void **)((char *)self + 0x250);
  if (cargoHolder == 0) return false;
  cargo = *(void **)((char *)cargoHolder + 0x2b0);
  if (cargo == 0) return false;
  locker = *(void **)((char *)self + 0xcbc);
  if (locker == 0) return false;
  if (*(int *)((char *)obj + 0x38) != 0x1a) return false;

  blob = *(int **)((char *)obj + 0x3c);
  if (blob == 0) return false;

  // Assembly rewrites stack mode slot to (mode != 2) for secondary gate later.
  modeNot2 = (mode != 2);

  // ---- simple path: count only blob+0x4c0 across cargo+locker ----
  if (simpleFlag != 0) {
    *outBatches  = InventoryGrid_CountItemsByCbid(cargo,  blob[0x4c0/4], 0);
    *outBatches += InventoryGrid_CountItemsByCbid(locker, blob[0x4c0/4], 0);
    return *outBatches != 0;
  }

  // ---- primary bank: blob+0x4d0 CBID, cost blob+0x4d4 ----
  if (mode != 1 && *(char *)((char *)blob + 0x4d4) > 0) {
    primaryHave  = InventoryGrid_CountItemsByCbid(cargo,  *(int *)((char *)blob + 0x4d0), 0);
    primaryHave += InventoryGrid_CountItemsByCbid(locker, *(int *)((char *)blob + 0x4d0), 0);
    if (primaryHave >= (int)*(char *)((char *)blob + 0x4d4)) {
      *outBatches += primaryHave / (int)*(char *)((char *)blob + 0x4d4);
    }
  }

  // ---- secondary list: min count across non-(-1) slots @ +0x498 ----
  if (modeNot2) {
    nSlots = ItemDef_CountConfiguredModSlots_Inferred(blob); // 0..5 non-(-1)
    if (nSlots > 0) {
      minHave = -1;
      for (i = 0; i < nSlots; ++i) {
        slotCbid = *(int *)((char *)blob + 0x498 + 4 * i);
        if (slotCbid != -1) {
          secondaryHave  = InventoryGrid_CountItemsByCbid(cargo,  slotCbid, 0);
          secondaryHave += InventoryGrid_CountItemsByCbid(locker, slotCbid, 0);
          if (minHave == -1 || secondaryHave <= minHave)
            minHave = secondaryHave;
          if (secondaryHave < 1)
            break; // odd: stop scan; min may be 0
        }
      }
      if (minHave > 0)
        *outBatches += minHave; // ADD not MIN with primary — sealed
    }
  }

  return *outBatches > 0;
}
```

## Control-flow map

```text
entry
  *out=0
  guard cargoHolder / cargo / locker / type==0x1a ──fail──► AL=0 RET 10
  blob = obj+0x3c ──null──► AL=0 RET 10
  modeNot2 = (mode != 2)
  if simpleFlag ≠ 0:
      dual-count blob+0x4c0 → *out; AL = (*out != 0); RET 10
  if mode ≠ 1 and cost@+0x4d4 > 0:
      dual-count primary@+0x4d0
      if have >= cost: *out += have/cost
  if modeNot2:
      n = count non-(-1) slots
      min over sequential slots[0..n)
      if min > 0: *out += min
  AL = (*out > 0); RET 10
```

---

## Odd behaviors preserved

1. **Primary batches + secondary min are ADDed** into `*outBatches` (not `min(primary, secondary)`). Parent craft consume uses the sum as the affordability ceiling.
2. Secondary loop iterates **indices `0 .. nSlots-1`** where `nSlots` is the **count** of non-`-1` slots, not a packed dense list — holes among the first `n` dwords are skipped (`!= -1`) but slots beyond `n` are never visited.
3. Simple path uses **`+0x4c0`**, not primary `+0x4d0`.
4. First cargo `CountItemsByCbid` often relies on **live ECX** from the guard load (no reload).
