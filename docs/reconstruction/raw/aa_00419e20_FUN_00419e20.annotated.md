/* Com_BstrHolder_Release
   ESI = holder* (12-byte BSTR holder).
   Returns: new refcount if still live; 0 after free. */

LONG Com_BstrHolder_Release(/* ESI */ ComBstrHolder *self)
{
  LONG newCount = InterlockedDecrement(&self->refcount); /* +8 */
  if (newCount != 0) {
    return self->refcount; /* re-read [esi+8] */
  }
  if (self != 0) {
    if (self->bstr != 0) {
      SysFreeString(self->bstr);
    }
    if (self->secondary != 0) {
      operator_delete[](self->secondary);
    }
    operator_delete(self); /* returns; then fall through */
  }
  return 0;
}
