// Returns XP remaining until soft-cap at max level (threshold(maxLevel) - totalXp - 1)
int Character_GetXpRemainingToMaxLevelSoftCap_Inferred(void /* ESI */)
{
  uint threshold;
  Character* character; // ESI

  // +0xc50 = nMaxLevel (ushort key into tExperienceLevel)
  threshold = Experience_GetCumulativeThreshold(*(ushort*)((char*)character + 0xc50));

}
