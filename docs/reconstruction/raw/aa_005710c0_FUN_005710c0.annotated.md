## Machine-level notes

- Source: raw capture for `aa_005710c0` + live re-decompile + `read_memory` 256 B.
- **Match key is CBID** (`clonebase+0x34`), not instance COID. Older “find-by-COID” labels are wrong.
- List subobject: `lea ebp, [ecx+0x2c]` before TraversalLock.
- Capacity compare zero-extends max to 16 bits (`movzx edx, si`) even after loading 999999.
- `ret 0x0C` → three stack args + thiscall ECX.

## Pseudocode (annotated)

```c
int * __thiscall InventoryGrid_FindItemByCbid(
    int grid /*this*/, int cbid, char skipStackGate, ushort addQty)
{
  ushort maxField;
  int *item;
  int clonebase;
  uint maxStack;
  int *iterOut;
  undefined4 iterCursor;

  iterCursor = 0;
  FUN_004294f0();                    // List_TraversalLock(grid+0x2C)
  int done = FUN_004022a0(&iterCursor, &iterOut); // IterateNext
  item = iterOut;
  do {
    if (done != 0) {
      // unlock grid+0x54 / CS grid+0x30
      return NULL;
    }
    iterOut = item;
    // item[0x2a] == item+0xA8 == clonebase*
    if ((item != NULL) && (clonebase = item[0x2a], *(int*)(clonebase + 0x34) == cbid)) {
      if (skipStackGate != 0) {
        // unlock; return item  (Broken allowed)
        return item;
      }
      // item[0x5f] == item+0x17C flags; bit19 Broken
      if (((uint)item[0x5f] >> 0x13 & 1) == 0) {
        if (*(int*)(clonebase + 0x38) == 4) {
          maxStack = 999999;         // then & 0xFFFF → 16959 in compare
        } else {
          maxField = *(ushort*)(*(int*)(clonebase + 0x3c) + 0x4ba);
          maxStack = maxField;
          if (maxField == 0) maxStack = 200;
        }
        int qty = (**(code**)(*item + 0x25c))();  // quantity
        if ((int)(qty + (uint)addQty) <= (int)(maxStack & 0xffff))
          return item; // after unlock
      }
    }
    done = FUN_004022a0(&iterCursor, &iterOut);
    item = iterOut;
  } while (true);
}
```
